cmake_minimum_required(VERSION 3.8)
project(v2_message_consumer_check)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rm_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_executable(sentry_command_mux_check "${ROOT}/src/rm_sentry_decision/src/sentry_command_mux_node.cpp")
ament_target_dependencies(sentry_command_mux_check rclcpp rm_interfaces sensor_msgs std_msgs)
add_executable(referee_check referee_check.cpp "${ROOT}/src/sentry_bt/src/referee_interface.cpp")
target_include_directories(referee_check PRIVATE "${ROOT}/src/sentry_bt/include")
ament_target_dependencies(referee_check rm_interfaces)
enable_testing()
add_test(NAME missing_referee_state COMMAND referee_check)
